cmake_minimum_required(VERSION 3.18)
project(robot_startup C)
set(CMAKE_C_STANDARD 11)
find_path(ROBOT_INCLUDE_DIR robot_hw.h REQUIRED)
find_library(ROBOT_LIBRARY NAMES robot REQUIRED)
add_executable(robot-hw-check robot-hw-check.c)
target_include_directories(robot-hw-check PRIVATE ${ROBOT_INCLUDE_DIR})
target_link_libraries(robot-hw-check PRIVATE ${ROBOT_LIBRARY} m)
target_compile_options(robot-hw-check PRIVATE -Wall -Wextra -Werror)
install(TARGETS robot-hw-check RUNTIME DESTINATION libexec/robot)
